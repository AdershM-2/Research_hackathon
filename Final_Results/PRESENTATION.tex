% PHYSICS PRINCIPLE
%================================================================================
\begin{frame}{System Architecture: Lorentz Force}
\begin{columns}[t]
\column{0.5\textwidth}
\begin{block}{Physics Principle}
\Large
$$\vec{F} = q(\vec{v} \times \vec{B})$$
\end{block}

\normalsize
\textbf{Parameters:}
\begin{itemize}
    \item $q$ = Charge (1-5 Coulombs)
    \item $v$ = Orbital velocity ($\sim$7.5 km/s)
    \item $B$ = Earth's magnetic field ($\sim$30-50 $\mu$T)
\end{itemize}

\vspace{0.3cm}
\textbf{Force Output:}
\begin{itemize}
    \item Range: 1-5 N
    \item Propellant mass: 0 kg
    \item Thrust duration: Unlimited
\end{itemize}

\column{0.5\textwidth}
\begin{block}{Configuration}
\textbf{6 Coulomb Shells:}
\begin{itemize}
    \item $\pm x$ direction (2 shells)
    \item $\pm y$ direction (2 shells)
    \item $\pm z$ direction (2 shells)
\end{itemize}
\end{block}

\textbf{Shell Separation:} 0.5-5 m tested
\begin{itemize}
    \item Larger separation: higher torque
    \item Smaller separation: compact design
\end{itemize}

\textbf{Charge Levels:}
\begin{itemize}
    \item 1 C/shell
    \item 3 C/shell
    \item 5 C/shell
\end{itemize}
\end{columns}
\end{frame}
