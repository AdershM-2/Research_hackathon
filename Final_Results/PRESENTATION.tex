\begin{frame}{System Architecture: Lorentz Force}
\begin{columns}[t]
\column{0.5\textwidth}
\begin{block}{Physics Principle}
\Large
$$\vec{F} = q(\vec{v} \times \vec{B})$$
\end{block}

\normalsize
\textbf{Where:}
\begin{itemize}
    \item $q$ = Charge (1-5 Coulombs)
    \item $v$ = Orbital velocity ($\sim$7.5 km/s)
    \item $B$ = Earth's magnetic field ($\sim$30-50 $\mu$T)
\end{itemize}

\vspace{0.3cm}
\textbf{Result:}
\begin{itemize}
    \item Force: 1-5 N
    \item No propellant needed
    \item Continuous thrust available
\end{itemize}

\column{0.5\textwidth}
\begin{block}{System Configuration}
\textbf{6 Coulomb Shells:}
\begin{itemize}
    \item $\pm x$ direction (2 shells)
    \item $\pm y$ direction (2 shells)
    \item $\pm z$ direction (2 shells)
\end{itemize}
\end{block}

\textbf{Shell Separation:} 0.5-5 m tested
\begin{itemize}
    \item Larger = higher torque
    \item Smaller = compact design
\end{itemize}

\textbf{Charge Levels:}
\begin{itemize}
    \item 1 C/shell (conservative)
    \item 3 C/shell (balanced)
    \item 5 C/shell (aggressive)
\end{itemize}
\end{columns}
\end{frame}
