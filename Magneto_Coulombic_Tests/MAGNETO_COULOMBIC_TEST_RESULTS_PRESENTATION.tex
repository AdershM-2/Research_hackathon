\section{System Overview}

\begin{frame}{Magneto-Coulombic Actuation Principle}
\begin{columns}
\column{0.5\textwidth}
\textbf{Core Physics: Lorentz Force}
\begin{equation}
\vec{F} = q(\vec{v} \times \vec{B})
\end{equation}

Where:
\begin{itemize}
    \item $q$ = charge (±1 μC per shell)
    \item $\vec{v}$ = orbital velocity (~7,670 m/s)
    \item $\vec{B}$ = Earth's B-field (~25 μT)
    \item $\vec{F}$ = Lorentz force (μN range)
\end{itemize}

\column{0.5\textwidth}
\textbf{System Configuration:}
\begin{itemize}
    \item 6 Coulomb shells
    \item Positions: ±1m along x,y,z
    \item Voltage: 50 kV
    \item Energy: 500 MJ capacity
    \item Mass: 1270 kg (dry)
\end{itemize}

\vspace{1em}
\textbf{Operating Modes:}
\begin{itemize}
    \item All shells +q → Net force
    \item Pairs ±q → Torque
\end{itemize}
\end{columns}
\end{frame}
